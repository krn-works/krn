import pytest

import humanize


@pytest.mark.parametrize(
    ("value", "expected"),
    [
        (1e16, "10,000,000,000,000,000.0"),
        (1.5e20, "150,000,000,000,000,000,000.0"),
        (1e22, "10,000,000,000,000,000,000,000.0"),
        (-1.5e16, "-15,000,000,000,000,000.0"),
        (0.00001, "0.00001"),
    ],
)
def test_no_exponent_notation(value, expected):
    assert humanize.intcomma(value) == expected


@pytest.mark.parametrize(
    ("args", "expected"),
    [
        ((1e15,), "1,000,000,000,000,000.0"),
        ((1234567.25,), "1,234,567.25"),
        ((1e16, 0), "10,000,000,000,000,000"),
        ((1000,), "1,000"),
    ],
)
def test_existing_behaviour(args, expected):
    assert humanize.intcomma(*args) == expected
